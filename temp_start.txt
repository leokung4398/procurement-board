      const list = [];
      snap.forEach(doc => {

        if (b.status !== 'deleted') {

  },
  async getMonths() {
    const list = await this.getAllBulletins();
    const m = {};
    list.forEach(b => {
      const mk = (b.publishDate || '').slice(0, 7) || '?';
      if (!m[mk]) m[mk] = [];
      m[mk].push({ id: b.id, status: b.status || 'published', sections: b.sections, publishDate: b.publishDate });
    });
    return m;
  },
